Algorithm TaggedTotal
INPUT: Record
OUTPUT: Amount
BEGIN
    SWITCH Record.Kind
        CASE OP: RETURN Record.Consultation + Record.Tests
        CASE IP: RETURN Record.Days × Record.Rate + Record.Procedures + Record.Medicines
        CASE DC: RETURN Record.PackagePrice
        CASE ER: RETURN Record.Base + Record.Hours × Record.HourlyRate
    END SWITCH
    RAISE UnknownBillType
END
