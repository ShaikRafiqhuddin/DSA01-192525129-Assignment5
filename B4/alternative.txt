Algorithm InterpretConfiguredBill
INPUT: BillData, VersionedRules
OUTPUT: Amount
BEGIN
    CALL ValidateRuleSchema(VersionedRules)
    Amount ← 0
    FOR EACH Rule IN VersionedRules[BillData.Kind] DO
        Amount ← Amount + CALL EvaluateAllowedFormula(Rule, BillData)
    END FOR
    RETURN CALL ValidateMoney(Amount)
END
